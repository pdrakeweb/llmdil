//! Per-backend memory requirements: how much GPU RAM and system
//! RAM a configured backend needs, given its model's weight layout
//! and its own serving flags (tensor split, expert offload, context
//! size).
//!
//! The weight split comes from the GGUF tensor directory when the
//! file is available ([`MemoryBasis::Exact`]), from the curated
//! catalog's pinned expert share before download
//! ([`MemoryBasis::Catalog`]), or is not known at all
//! ([`MemoryBasis::Estimate`] — the requirement then degrades to
//! the pre-offload behavior: everything counted on the GPU).

use serde::{Deserialize, Serialize};

use super::MIB;

/// How a [`MemoryRequirement`] was derived.
#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
#[serde(rename_all = "snake_case")]
pub enum MemoryBasis {
    /// Parsed from the downloaded GGUF's tensor directory.
    Exact,
    /// From the curated catalog's pinned sizes.
    Catalog,
    /// Estimated (parameter-count derivation, or an offload split
    /// that could not be resolved). Figures are approximate.
    Estimate,
}

/// Expert offload configured in a backend's raw llama-server flags.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ExpertOffload {
    /// No offload: all weights are GPU-resident (subject to the
    /// tensor split).
    None,
    /// All routed experts on the CPU (`-ot exps=CPU` and spellings
    /// of the same override, or `--cpu-moe` / `-cmoe`).
    AllExperts,
    /// Experts of the first N layers on the CPU (`--n-cpu-moe N`,
    /// `-ncmoe N`).
    CpuMoeLayers(u32),
}

impl ExpertOffload {
    /// The offload slider position this setting represents: how
    /// many of the model's `moe_layers` MoE layers sit in system
    /// RAM. A partial count past the model's layer count clamps
    /// (llama.cpp treats `--n-cpu-moe` past the end as all).
    pub fn layers(&self, moe_layers: u32) -> u32 {
        match self {
            ExpertOffload::None => 0,
            ExpertOffload::AllExperts => moe_layers,
            ExpertOffload::CpuMoeLayers(n) => (*n).min(moe_layers),
        }
    }
}

/// The offload setting a slider position selects: 0 layers is no
/// offload, the full MoE layer count is all experts (written in
/// the `-ot exps=CPU` form), anything between is
/// `--n-cpu-moe N`.
pub fn offload_for_layers(layers: u32, moe_layers: u32) -> ExpertOffload {
    if layers == 0 || moe_layers == 0 {
        ExpertOffload::None
    } else if layers >= moe_layers {
        ExpertOffload::AllExperts
    } else {
        ExpertOffload::CpuMoeLayers(layers)
    }
}

/// The canonical raw-flags text for an offload setting — the form
/// the app itself writes. `None` when the setting writes no tokens
/// (no offload).
pub fn offload_flag_string(offload: ExpertOffload) -> Option<String> {
    match offload {
        ExpertOffload::None => None,
        ExpertOffload::AllExperts => Some("-ot exps=CPU".to_string()),
        ExpertOffload::CpuMoeLayers(0) => None,
        ExpertOffload::CpuMoeLayers(n) => Some(format!("--n-cpu-moe {n}")),
    }
}

/// The offload slider's color-band boundaries as fractions of the
/// offloaded range: `(green_end, yellow_end)`. Green runs from no
/// offload to the model's green limit — the catalog's default
/// offload, the most offload that is still fast for the model.
/// Yellow covers the first half of the range past the limit and
/// red the rest, so the bands derive from the same catalog number
/// the install default uses and the two can never disagree.
pub fn offload_bands(moe_layers: u32, green_limit_layers: u32) -> (f64, f64) {
    if moe_layers == 0 {
        return (0.0, 0.0);
    }
    let green = (green_limit_layers.min(moe_layers) as f64 / moe_layers as f64).clamp(0.0, 1.0);
    let yellow = green + (1.0 - green) / 2.0;
    (green, yellow)
}

/// Whether one comma-separated `--override-tensor` assignment
/// targets routed experts (its pattern mentions `exps`, as in
/// `exps=CPU` or `.*_exps.*=CPU`) — the assignments expert
/// offload owns. Mirrors [`expert_offload`]'s detection.
fn override_part_is_expert(part: &str) -> bool {
    part.split('=').next().unwrap_or("").contains("exps")
}

/// Rewrite the expert-offload tokens in raw llama-server flags,
/// leaving every other flag untouched and in order. All existing
/// offload spellings are removed first (`--n-cpu-moe` / `-ncmoe`
/// in both forms, `--cpu-moe` / `-cmoe`, and any `-ot` /
/// `--override-tensor` assignment whose pattern mentions `exps` —
/// sibling assignments in the same `-ot` list keep their flag),
/// then the canonical tokens for `offload` are appended. Returns
/// `None` when nothing remains.
pub fn set_expert_offload(raw_flags: Option<&str>, offload: ExpertOffload) -> Option<String> {
    let tokens: Vec<&str> = raw_flags.unwrap_or("").split_whitespace().collect();
    let mut out: Vec<String> = Vec::new();
    let mut i = 0;
    while i < tokens.len() {
        let t = tokens[i];
        if t == "--n-cpu-moe" || t == "-ncmoe" {
            // Skip the value token too when it parses as a count;
            // a malformed value is left for the main walk.
            let has_value = tokens
                .get(i + 1)
                .map(|v| v.parse::<u32>().is_ok())
                .unwrap_or(false);
            i += if has_value { 2 } else { 1 };
            continue;
        }
        if t.starts_with("--n-cpu-moe=") || t.starts_with("-ncmoe=") {
            i += 1;
            continue;
        }
        if t == "--cpu-moe" || t == "-cmoe" {
            // Valueless all-experts spelling: drop the token alone.
            i += 1;
            continue;
        }
        let (flag, inline): (&str, Option<&str>) = if t == "-ot" || t == "--override-tensor" {
            (t, None)
        } else if let Some(v) = t.strip_prefix("--override-tensor=") {
            ("--override-tensor", Some(v))
        } else if let Some(v) = t.strip_prefix("-ot=") {
            ("-ot", Some(v))
        } else {
            out.push(t.to_string());
            i += 1;
            continue;
        };
        let value: Option<&str> = match inline {
            Some(v) => {
                i += 1;
                Some(v)
            }
            None => {
                let v = tokens.get(i + 1).copied();
                i += if v.is_some() { 2 } else { 1 };
                v
            }
        };
        let Some(value) = value else {
            // A trailing override flag with no value is malformed
            // but not ours to delete.
            out.push(flag.to_string());
            continue;
        };
        let kept: Vec<&str> = value
            .split(',')
            .filter(|part| !override_part_is_expert(part))
            .collect();
        if !kept.is_empty() {
            out.push(flag.to_string());
            out.push(kept.join(","));
        }
    }
    match offload {
        ExpertOffload::AllExperts => {
            out.push("-ot".to_string());
            out.push("exps=CPU".to_string());
        }
        ExpertOffload::CpuMoeLayers(n) if n > 0 => {
            out.push("--n-cpu-moe".to_string());
            out.push(n.to_string());
        }
        _ => {}
    }
    if out.is_empty() {
        None
    } else {
        Some(out.join(" "))
    }
}

/// Detect expert offload in raw llama-server flags. Recognizes
/// `--override-tensor` / `-ot` assignments whose pattern mentions
/// `exps` with a CPU buffer (the `-ot exps=CPU` form the app itself
/// writes, plus `=` spellings), `--cpu-moe` / `-cmoe` (all experts
/// on the CPU), and `--n-cpu-moe N` / `-ncmoe N` (first N layers).
///
/// When both an all-experts spelling and a layer count are present
/// the all-experts setting wins regardless of order: llama.cpp
/// applies both, and `-ot exps=CPU` / `--cpu-moe` already put every
/// expert tensor on the CPU, leaving nothing for the layer count
/// to move. A repeated layer count resolves to the last one, the
/// same sequential-parse rule llama.cpp follows.
pub fn expert_offload(raw_flags: Option<&str>) -> ExpertOffload {
    let Some(raw) = raw_flags else {
        return ExpertOffload::None;
    };
    let tokens: Vec<&str> = raw.split_whitespace().collect();
    let mut layers: Option<u32> = None;
    let mut i = 0;
    while i < tokens.len() {
        let t = tokens[i];
        if t == "--cpu-moe" || t == "-cmoe" {
            return ExpertOffload::AllExperts;
        }
        let assignment = if t == "-ot" || t == "--override-tensor" {
            i += 1;
            tokens.get(i).copied()
        } else {
            t.strip_prefix("--override-tensor=")
                .or_else(|| t.strip_prefix("-ot="))
        };
        if let Some(list) = assignment {
            for part in list.split(',') {
                if let Some((pattern, buffer)) = part.split_once('=') {
                    if pattern.contains("exps") && buffer.eq_ignore_ascii_case("CPU") {
                        return ExpertOffload::AllExperts;
                    }
                }
            }
        }
        if t == "--n-cpu-moe" || t == "-ncmoe" {
            if let Some(v) = tokens.get(i + 1) {
                if let Ok(n) = v.parse::<u32>() {
                    if n > 0 {
                        layers = Some(n);
                    }
                }
            }
        } else if let Some(v) = t
            .strip_prefix("--n-cpu-moe=")
            .or_else(|| t.strip_prefix("-ncmoe="))
        {
            if let Ok(n) = v.parse::<u32>() {
                if n > 0 {
                    layers = Some(n);
                }
            }
        }
        i += 1;
    }
    match layers {
        Some(n) => ExpertOffload::CpuMoeLayers(n),
        None => ExpertOffload::None,
    }
}

/// One backend's memory requirement under its current configuration.
#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
pub struct MemoryRequirement {
    /// Model weights resident on the GPU (all GPUs combined when a
    /// tensor split divides them; per-GPU shares are the placement
    /// projection's job).
    pub gpu_weights_mib: u64,
    /// Model weights resident in system RAM (offloaded experts).
    pub sys_weights_mib: u64,
    /// KV cache at the backend's configured context size, when the
    /// model's KV geometry is known. Weights only otherwise.
    pub kv_mib: Option<u64>,
    /// Recurrent-state cache (llama.cpp's "rs cache") at the
    /// backend's effective slot count, when the model's
    /// recurrent-state geometry is known (hybrid SSM / delta-net
    /// architectures). The state is per sequence and independent
    /// of context size; dense models carry none. Absent when the
    /// geometry is unknown — never counted as zero.
    pub rs_mib: Option<u64>,
    /// MoE expert cache (`--moe-cache-mib`): the TOTAL VRAM
    /// budget the backend reserves for caching host-resident
    /// experts on the GPU, across all placed GPUs (llama.cpp
    /// splits the one budget among the devices holding offloaded
    /// layers — see [`super::moe_cache::moe_cache_shares`] for
    /// the per-GPU distribution consumers price). 0 when the
    /// backend sets no cache. Set by the requirement builders
    /// from the decoded flags, not by [`memory_requirement`]
    /// (which prices weights/KV/state and never sees the flag).
    #[serde(default)]
    pub moe_cache_mib: u64,
    /// The backend's resolved offload position (MoE layers in
    /// system RAM) at requirement time — the distribution input
    /// [`super::moe_cache::moe_cache_shares`] needs alongside
    /// [`Self::moe_cache_moe_layers`]. 0 when unknown or no
    /// offload.
    #[serde(default)]
    pub moe_cache_offload_layers: u32,
    /// The model's MoE layer count at requirement time (the
    /// distribution's layer-range basis). 0 when unknown.
    #[serde(default)]
    pub moe_cache_moe_layers: u32,
    pub basis: MemoryBasis,
}

fn mib(bytes: u64) -> u64 {
    bytes.div_ceil(MIB)
}

/// Compute a backend's memory requirement.
///
/// `expert_bytes` is the model's routed-expert total (from the GGUF
/// layout or the catalog pin); `moe_layers` — the number of blocks
/// that carry routed experts — is only needed to resolve a partial
/// [`ExpertOffload::CpuMoeLayers`] share. It must be the MoE layer
/// count, not the total block count: `--n-cpu-moe N` counts MoE
/// layers, and hybrid models carry experts in only some blocks
/// (Nemotron: 24 of 53).
/// `kv_bytes_per_token` comes from the GGUF KV geometry.
///
/// `rs_bytes_per_seq` is the model's recurrent-state bytes for
/// one sequence (from the GGUF recurrent-state geometry) and
/// `n_seq` the backend's effective slot count: the state cache
/// llama.cpp allocates is the per-sequence figure times the slot
/// count, independent of `n_ctx`. (llama.cpp can add rollback
/// snapshot planes for speculative state restore; those are not
/// modeled — the term prices the live state, one row per slot.)
///
/// `expert_bytes_by_block` carries the per-MoE-block expert sums
/// from the GGUF layout (in block order), when available. A partial
/// layer offload moves the *first* N MoE blocks, whose byte total
/// is generally not the uniform `N / layers` share — Unsloth
/// Dynamic quants deliberately quantize different layers at
/// different rates. With per-block sums the partial share is the
/// true sum of the first N blocks and an [`MemoryBasis::Exact`]
/// input stays exact; without them the share falls back to the
/// uniform proportion and an `Exact` input is downgraded to
/// [`MemoryBasis::Estimate`] rather than asserting precision the
/// numbers do not have (a [`MemoryBasis::Catalog`] input stays
/// `Catalog` — the catalog pins only aggregates by design).
///
/// Degradation rule: when offload is configured but the expert
/// share cannot be resolved (unknown expert bytes, or a partial
/// layer offload without a MoE layer count), the split is
/// unresolvable and the requirement falls back to
/// all-weights-on-GPU with basis [`MemoryBasis::Estimate`] — never
/// a silently invented split.
// The parameter list is the layout itself (totals, expert shares,
// basis, offload, KV geometry, context) — callers pass a resolved
// layout's fields positionally, as the other multi-argument entry
// points in this crate do.
#[allow(clippy::too_many_arguments)]
pub fn memory_requirement(
    total_bytes: u64,
    expert_bytes: Option<u64>,
    expert_bytes_by_block: Option<&[u64]>,
    basis: MemoryBasis,
    offload: ExpertOffload,
    kv_bytes_per_token: Option<u64>,
    moe_layers: Option<u32>,
    n_ctx: u32,
    rs_bytes_per_seq: Option<u64>,
    n_seq: u32,
) -> MemoryRequirement {
    let kv_mib = kv_bytes_per_token.map(|per| mib(per.saturating_mul(n_ctx as u64)));
    let rs_mib = rs_bytes_per_seq
        .map(|per| mib(per.saturating_mul(n_seq.max(1) as u64)));
    let mut basis = basis;
    let sys_bytes: u64 = match offload {
        ExpertOffload::None => 0,
        ExpertOffload::AllExperts => match expert_bytes {
            Some(e) => e.min(total_bytes),
            None => {
                return MemoryRequirement {
                    gpu_weights_mib: mib(total_bytes),
                    sys_weights_mib: 0,
                    kv_mib,
                    rs_mib,
                    moe_cache_mib: 0,
                    moe_cache_offload_layers: 0,
                    moe_cache_moe_layers: 0,
                    basis: MemoryBasis::Estimate,
                }
            }
        },
        ExpertOffload::CpuMoeLayers(n) => match (expert_bytes, moe_layers) {
            (Some(e), Some(layers)) if layers > 0 => {
                let share = n.min(layers);
                match expert_bytes_by_block {
                    // The first `share` MoE blocks' true byte sum.
                    Some(blocks) if blocks.len() == layers as usize => {
                        blocks[..share as usize].iter().sum::<u64>()
                    }
                    _ => {
                        // Uniform-share approximation: exact only
                        // at the endpoints (nothing or everything
                        // offloaded); a genuinely partial share
                        // cannot carry an Exact basis.
                        if share > 0 && share < layers && basis == MemoryBasis::Exact {
                            basis = MemoryBasis::Estimate;
                        }
                        (e.min(total_bytes) * share as u64) / layers as u64
                    }
                }
            }
            _ => {
                return MemoryRequirement {
                    gpu_weights_mib: mib(total_bytes),
                    sys_weights_mib: 0,
                    kv_mib,
                    rs_mib,
                    moe_cache_mib: 0,
                    moe_cache_offload_layers: 0,
                    moe_cache_moe_layers: 0,
                    basis: MemoryBasis::Estimate,
                }
            }
        },
    };
    MemoryRequirement {
        gpu_weights_mib: mib(total_bytes - sys_bytes.min(total_bytes)),
        sys_weights_mib: mib(sys_bytes),
        kv_mib,
        rs_mib,
        moe_cache_mib: 0,
        moe_cache_offload_layers: 0,
        moe_cache_moe_layers: 0,
        basis,
    }
}

/// System-RAM weight bytes for every offload position
/// `0..=moe_layers`: entry `n` is the byte total
/// [`memory_requirement`] assigns to system RAM when the first
/// `n` MoE layers' experts are offloaded
/// ([`ExpertOffload::CpuMoeLayers`], resolved through
/// [`offload_for_layers`] at the endpoints). Position 0 is 0 and
/// position `moe_layers` is `expert_bytes`.
///
/// With per-block sums whose length matches `moe_layers`, entry
/// `n` is the true prefix sum of the first `n` blocks — the same
/// arithmetic `memory_requirement`'s `CpuMoeLayers` branch uses.
/// Without them (or with a mismatched list) it is the uniform
/// share `expert_bytes * n / moe_layers`, again mirroring that
/// branch. Callers that render a per-position table (the offload
/// slider's RAM figures, guardrail evaluation) read it from here
/// so the numbers can never drift from the requirement itself.
pub fn sys_bytes_by_position(
    expert_bytes: u64,
    expert_bytes_by_block: Option<&[u64]>,
    moe_layers: u32,
) -> Vec<u64> {
    if moe_layers == 0 {
        return vec![0];
    }
    let blocks = match expert_bytes_by_block {
        Some(blocks) if blocks.len() == moe_layers as usize => Some(blocks),
        _ => None,
    };
    (0..=moe_layers)
        .map(|n| match blocks {
            Some(blocks) => blocks[..n as usize].iter().sum(),
            None => expert_bytes * n as u64 / moe_layers as u64,
        })
        .collect()
}

#[cfg(test)]
mod tests {
    use super::*;

    const GIB: u64 = 1024 * MIB;

    #[test]
    fn detects_offload_spellings() {
        assert_eq!(
            expert_offload(Some("-ot exps=CPU")),
            ExpertOffload::AllExperts
        );
        assert_eq!(
            expert_offload(Some("--override-tensor exps=CPU")),
            ExpertOffload::AllExperts
        );
        assert_eq!(
            expert_offload(Some("--override-tensor=.*_exps.*=CPU")),
            ExpertOffload::AllExperts
        );
        assert_eq!(
            expert_offload(Some("--threads 8 -ot ffn=CUDA0")),
            ExpertOffload::None
        );
        assert_eq!(
            expert_offload(Some("--n-cpu-moe 24")),
            ExpertOffload::CpuMoeLayers(24)
        );
        assert_eq!(expert_offload(Some("--n-cpu-moe=0")), ExpertOffload::None);
        assert_eq!(expert_offload(Some("")), ExpertOffload::None);
        assert_eq!(expert_offload(None), ExpertOffload::None);
    }

    #[test]
    fn detects_cpu_moe_spellings() {
        // llama.cpp's other spellings: --cpu-moe / -cmoe put ALL
        // experts on the CPU; -ncmoe is --n-cpu-moe's short form.
        assert_eq!(expert_offload(Some("--cpu-moe")), ExpertOffload::AllExperts);
        assert_eq!(expert_offload(Some("-cmoe")), ExpertOffload::AllExperts);
        assert_eq!(
            expert_offload(Some("--threads 8 --cpu-moe")),
            ExpertOffload::AllExperts
        );
        assert_eq!(
            expert_offload(Some("-ncmoe 12")),
            ExpertOffload::CpuMoeLayers(12)
        );
        assert_eq!(
            expert_offload(Some("-ncmoe=12")),
            ExpertOffload::CpuMoeLayers(12)
        );
        // Both spellings present: all-experts wins in either order,
        // because the process applies both and -ot/--cpu-moe leave
        // nothing for the layer count to move.
        assert_eq!(
            expert_offload(Some("--n-cpu-moe 3 -ot exps=CPU")),
            ExpertOffload::AllExperts
        );
        assert_eq!(
            expert_offload(Some("-ot exps=CPU --n-cpu-moe 3")),
            ExpertOffload::AllExperts
        );
        assert_eq!(
            expert_offload(Some("--n-cpu-moe 3 --cpu-moe")),
            ExpertOffload::AllExperts
        );
        // A repeated layer count resolves to the last one.
        assert_eq!(
            expert_offload(Some("--n-cpu-moe 3 --n-cpu-moe 9")),
            ExpertOffload::CpuMoeLayers(9)
        );
    }

    #[test]
    fn rewrite_strips_cpu_moe_spellings() {
        // The surviving --cpu-moe would win in llama.cpp over the
        // slider's --n-cpu-moe, so every spelling must be removed
        // by the rewrite, in both directions.
        assert_eq!(
            set_expert_offload(Some("--cpu-moe"), ExpertOffload::CpuMoeLayers(8)).as_deref(),
            Some("--n-cpu-moe 8")
        );
        assert_eq!(
            set_expert_offload(Some("-cmoe --threads 4"), ExpertOffload::None).as_deref(),
            Some("--threads 4")
        );
        assert_eq!(
            set_expert_offload(Some("-ncmoe 5"), ExpertOffload::AllExperts).as_deref(),
            Some("-ot exps=CPU")
        );
        assert_eq!(
            set_expert_offload(Some("-ncmoe=5"), ExpertOffload::CpuMoeLayers(2)).as_deref(),
            Some("--n-cpu-moe 2")
        );
        // Round-trip: the detector reads back the target for
        // starts written in the new spellings too.
        for start in [Some("--cpu-moe"), Some("-ncmoe 7"), Some("-cmoe")] {
            for target in [
                ExpertOffload::None,
                ExpertOffload::CpuMoeLayers(13),
                ExpertOffload::AllExperts,
            ] {
                let out = set_expert_offload(start, target);
                assert_eq!(
                    expert_offload(out.as_deref()),
                    target,
                    "start {start:?} -> {target:?}"
                );
            }
        }
    }

    #[test]
    fn all_experts_offload_splits_weights() {
        // Coder-Next's real layout, rounded: 49 GB total, 46.7 GB experts.
        let r = memory_requirement(
            49 * GIB,
            Some(46 * GIB),
            None,
            MemoryBasis::Exact,
            ExpertOffload::AllExperts,
            Some(24_576),
            Some(48),
            32_768,
            None,
            4,
        );
        assert_eq!(r.sys_weights_mib, 46 * 1024);
        assert_eq!(r.gpu_weights_mib, 3 * 1024);
        // 24,576 B/token x 32,768 tokens = 768 MiB.
        assert_eq!(r.kv_mib, Some(768));
        assert_eq!(r.basis, MemoryBasis::Exact);
        // No recurrent-state geometry was passed: no state term.
        assert_eq!(r.rs_mib, None);
    }

    #[test]
    fn recurrent_state_term_completes_the_planner_moe_requirement() {
        // The planner-moe incident shape (Oct 20, 2026), in MiB:
        // the Whittle-MoE-27B Q8_0 file (27,350 MiB of tensors),
        // every routed expert on the CPU, 13,814 MiB of non-expert
        // weights left on the GPU, KV 2,048 MiB at n_ctx 32768
        // (65,536 B/token), and the recurrent-state geometry the
        // GGUF parse yields for the family (156,893,184 B per
        // sequence) at the auto slot count of 4. Before the rs
        // term existed this requirement summed to 15,862 MiB and
        // the start gate passed a configuration whose true total
        // — 16,460 MiB by llama.cpp's own allocation figures
        // (13,813.54 weights + 598.50 state + 2,048 KV) — exceeds
        // a 16,384 MiB card even empty.
        let mib_bytes = |m: u64| m * MIB;
        let r = memory_requirement(
            mib_bytes(27_350),
            Some(mib_bytes(13_536)),
            None,
            MemoryBasis::Exact,
            ExpertOffload::AllExperts,

//! Validation warnings for hand-edited raw llama-server flags
//! (tracker #60, the advanced flags editor).
//!
//! The editor's rule is warn, never block: any string saves, and
//! these warnings are how the app says what it noticed. The check
//! tokenizes with the same whitespace split every flag decoder in
//! this crate uses ([`super::memory::expert_offload`],
//! [`super::mtp::mtp_draft`]) and judges tokens against the flag
//! set the codebase itself recognizes — the flags its decoders
//! own, the flags its launch builder emits or interprets, and the
//! placement/GPU flags its placement code decodes. A flag outside
//! that set is reported as unrecognized, NOT as invalid: llama.cpp
//! knows more flags than this app models, and a hand editor exists
//! precisely for those. The UI mirrors this function for live
//! feedback while typing; this implementation is the authority
//! (the save op returns its result).

use serde::Serialize;

/// One validation warning about a raw-flags string. `token` is
/// the flag (or bare token) the warning is about, spelled as the
/// operator wrote it; `message` is the full warning text.
#[derive(Debug, Clone, PartialEq, Eq, Serialize)]
pub struct FlagWarning {
    pub token: String,
    pub message: String,
}

/// How a known flag takes its value.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Arity {
    /// No value (`--cpu-moe`, `--jinja`).
    None,
    /// The next token (or an `=` suffix) is the flag's value.
    Value(ValueKind),
    /// Like `Value`, but the flag may repeat without the
    /// duplicates warning (llama.cpp accumulates repeated
    /// `-ot` assignments).
    MultiValue(ValueKind),
}

/// The value shape a flag's value is checked against. Only the
/// shapes the app itself relies on are checked; everything else
/// is `Any` (presence is all the app can judge).
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum ValueKind {
    /// Must parse as an integer (`--parallel`, `--n-cpu-moe`).
    Int,
    /// Any single token.
    Any,
}

/// One recognized flag: its canonical name (duplicates are
/// judged across spellings), whether it is app-reserved (the
/// launch builder filters it out, so it never reaches
/// llama-server), and its arity.
struct FlagSpec {
    canonical: &'static str,
    reserved: bool,
    arity: Arity,
}

/// Look a token's flag name up in the recognized set. Aliases
/// map to one canonical spec so `--parallel` and `-np` count as
/// the same flag for duplicates.

        ("--cpu-moe", &FlagSpec { canonical: "--cpu-moe", reserved: false, arity: None }),
        ("-cmoe", &FlagSpec { canonical: "--cpu-moe", reserved: false, arity: None }),
        ("--override-tensor", &FlagSpec { canonical: "--override-tensor", reserved: false, arity: MultiValue(Any) }),
        ("-ot", &FlagSpec { canonical: "--override-tensor", reserved: false, arity: MultiValue(Any) }),
        ("--cache-type-k", &FlagSpec { canonical: "--cache-type-k", reserved: false, arity: Value(Any) }),
        ("-ctk", &FlagSpec { canonical: "--cache-type-k", reserved: false, arity: Value(Any) }),

/// Warnings for a raw-flags string, in token order. An empty or
/// absent string is valid and warns about nothing. Duplicate
/// and value warnings follow llama.cpp's sequential-parse rule:
/// for a repeated single-value flag the LAST occurrence is the
/// one that takes effect, which is what the warning says.
pub fn raw_flag_warnings(raw_flags: Option<&str>) -> Vec<FlagWarning> {
    let Some(raw) = raw_flags else {
        return Vec::new();
    };
    let tokens: Vec<&str> = raw.split_whitespace().collect();
    let mut warnings: Vec<FlagWarning> = Vec::new();
    let mut seen: Vec<&'static str> = Vec::new();
    let mut i = 0;
    while i < tokens.len() {
        let t = tokens[i];
        if !t.starts_with('-') || t == "-" {
            // A bare token the walk did not consume as a value.
            warnings.push(FlagWarning {
                token: t.to_string(),
                message: format!(
                    "`{t}` is not a flag and no flag takes it as a value; \
                     llama-server will reject it"
                ),
            });
            i += 1;
            continue;
        }
        // `--jinja=false` is the launch builder's third filtered
        // no-jinja spelling; judge it as that flag, not as --jinja
        // carrying a value.
        let (name, inline_value) = if t == "--jinja=false" {
            ("--no-jinja", None)
        } else {
            match t.split_once('=') {
                Some((n, v)) => (n, Some(v)),
                None => (t, None),
            }
        };
        let Some(spec) = flag_spec(name) else {
            warnings.push(FlagWarning {
                token: t.to_string(),
                message: format!(
                    "`{name}` is not a flag this app recognizes; \
                     it is passed to llama-server unchanged"
                ),
            });
            // An unrecognized flag's arity is unknowable; assume
            // the common one-value shape so its value is not
            // misreported as a stray token.
            if inline_value.is_none()
                && tokens
                    .get(i + 1)
                    .map(|v| !v.starts_with('-'))
                    .unwrap_or(false)
            {
                i += 1;
            }
            i += 1;
            continue;
        };
        if spec.reserved {
            warnings.push(FlagWarning {
                token: t.to_string(),
                message: format!(
                    "`{name}` is reserved: the app filters it out before launch, \
                     so it has no effect here"
                ),
            });
        }
        match spec.arity {
            Arity::None => {
                if inline_value.is_some() {
                    warnings.push(FlagWarning {
                        token: t.to_string(),
                        message: format!("`{name}` takes no value"),
                    });
                }
                if seen.contains(&spec.canonical) {
                    warnings.push(FlagWarning {
                        token: t.to_string(),
                        message: format!("`{}` is repeated", spec.canonical),
                    });
                }
                seen.push(spec.canonical);
                i += 1;
            }
            Arity::Value(kind) | Arity::MultiValue(kind) => {
                if seen.contains(&spec.canonical) && matches!(spec.arity, Arity::Value(_)) {
                    warnings.push(FlagWarning {
                        token: t.to_string(),
                        message: format!(
                            "`{}` is set more than once; the last value wins",
                            spec.canonical
                        ),
                    });
                }
                seen.push(spec.canonical);
                let value = match inline_value {
                    Some(v) => {
                        i += 1;
                        Some(v)
                    }
                    None => {
                        let v = tokens.get(i + 1).copied();
                        i += if v.is_some() { 2 } else { 1 };
                        v
                    }
                };
                match value {
                    None => warnings.push(FlagWarning {
                        token: t.to_string(),
                        message: format!("`{name}` has no value"),
                    }),
                    Some(v) => {
                        if kind == ValueKind::Int && v.parse::<i64>().is_err() {
                            warnings.push(FlagWarning {
                                token: t.to_string(),
                                message: format!(
                                    "`{name}` expects a number; `{v}` is not one"
                                ),
                            });
                        }
                    }
                }
            }
        }
    }
    warnings
}

//! KV cache types and flash attention in a backend's raw
//! llama-server flags: the `--cache-type-k` / `--cache-type-v`
//! element types the KV cache is stored in, and the
//! `--flash-attn` mode.
//!
//! The decode/encode pair mirrors the MTP pair in [`super::mtp`]
//! and the expert-offload pair in [`super::memory`]: state is
//! decoded from the saved raw flags — there is no separate stored
//! setting — and the writer rewrites only the tokens the setting
//! owns, leaving every other flag untouched and in order.
//! Malformed or hand-edited spellings degrade to a safe decoded
//! state, never a panic: an unrecognized cache type reads as the
//! llama.cpp default (f16, the value an absent token runs), and a
//! valueless flag the writer does not own is preserved rather
//! than deleted.
//!
//! The value sets are the pinned llama.cpp build's (b11658,
//! `common/arg.cpp`): `--cache-type-k/v` accept f32, f16, bf16,
//! q8_0, q4_0, q4_1, iq4_nl, q5_0, q5_1; this app offers the three
//! types v1 covers (f16, q8_0, q4_0) — [`CacheType`] is the clean
//! extension point for the rest. `--flash-attn` is tri-state on
//! the pinned build (`-fa, --flash-attn [on|off|auto]`, default
//! auto; `include/llama.h`'s `llama_flash_attn_type`), so the
//! control offers all three states.
//!
//! Two pinned-build rules shape what a valid selection is:
//! - A quantized V cache requires flash attention to be enabled
//!   (`src/llama-context.cpp`): with `--flash-attn auto` the
//!   server promotes it to enabled, with `off` context creation
//!   fails outright. [`CacheSelection::v_quantized_fa_off`] names
//!   the invalid combination so the save path can refuse it
//!   instead of persisting a backend that cannot start.
//! - Flash attention itself is supported on every architecture
//!   the pinned CUDA build targets, Pascal (sm_60) included: the
//!   CUDA backend's `ggml_cuda_flash_attn_ext_supported` falls
//!   back to a generic tile kernel when no tensor-core (Volta+)
//!   MMA kernel applies (`ggml/src/ggml-cuda/fattn.cu`), and
//!   q8_0/q4_0 are in its supported KV type set. There is
//!   therefore no compute-capability gate on the control — an
//!   earlier expectation that Pascal would need one does not
//!   hold against the pinned source.

use serde::{Deserialize, Serialize};

use super::gguf::KvGeometry;
use super::MIB;

/// One KV cache element type the app offers.
///
/// The byte geometry is ggml's block geometry for the type
/// (pinned llama.cpp `ggml/src/ggml-common.h`, mirrored by
/// [`super::gguf`]'s tensor type table): a block of
/// `block_elems` elements occupies `block_bytes` bytes, so an
/// element costs `block_bytes / block_elems` bytes — f16: 2/1,
/// q8_0: 34/32 (an fp16 scale plus 32 int8 quants), q4_0: 18/32
/// (an fp16 scale plus 32 nibbles). Pricing divides per cache
/// row (one head's vector for one token), rounding a partial
/// block up, so a head dimension that is not block-aligned is
/// overpriced by less than one block per row, never underpriced.
#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
#[serde(rename_all = "snake_case")]
pub enum CacheType {
    F16,
    Q8_0,
    Q4_0,
}

impl CacheType {
    /// Every type the control offers, in quality order.
    pub const ALL: [CacheType; 3] = [CacheType::F16, CacheType::Q8_0, CacheType::Q4_0];

    /// The llama.cpp spelling (`--cache-type-k q8_0`).
    pub fn as_str(&self) -> &'static str {
        match self {
            CacheType::F16 => "f16",
            CacheType::Q8_0 => "q8_0",
            CacheType::Q4_0 => "q4_0",
        }
    }

    /// Parse a llama.cpp spelling. Any other value — including
    /// types the pinned build accepts but v1 does not offer
    /// (q5_0, f32, ...) — reads as `None`: the decoder treats it
    /// as unset rather than inventing a position the control
    /// cannot render.
    pub fn parse(s: &str) -> Option<CacheType> {
        match s {
            "f16" => Some(CacheType::F16),
            "q8_0" => Some(CacheType::Q8_0),
            "q4_0" => Some(CacheType::Q4_0),
            _ => None,
        }
    }

    /// ggml block geometry: (elements per block, bytes per block).
    pub fn block_geometry(&self) -> (u64, u64) {
        match self {
            CacheType::F16 => (1, 2),
            CacheType::Q8_0 => (32, 34),
            CacheType::Q4_0 => (32, 18),
        }
    }

    /// Whether llama.cpp treats the type as quantized — the
    /// property the quantized-V rule keys on (f16 is not).
    pub fn is_quantized(&self) -> bool {
        !matches!(self, CacheType::F16)
    }

    /// Bytes for one cache row of `elems` elements at this type.
    pub fn row_bytes(&self, elems: u64) -> u64 {
        let (block_elems, block_bytes) = self.block_geometry();
        elems
            .div_ceil(block_elems)
            .saturating_mul(block_bytes)
    }
}

/// Flash attention mode in a backend's raw flags — the pinned
/// build's tri-state (`--flash-attn auto|on|off`).
#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
#[serde(rename_all = "snake_case")]
pub enum FlashAttn {
    Auto,
    On,
    Off,
}

impl FlashAttn {
    /// The llama.cpp spelling (`--flash-attn auto`).
    pub fn as_str(&self) -> &'static str {
        match self {
            FlashAttn::Auto => "auto",
            FlashAttn::On => "on",
            FlashAttn::Off => "off",
        }
    }

    /// Parse a llama.cpp spelling. The pinned build also accepts
    /// truthy/falsey spellings (1/true/yes, 0/false/no) through
    /// `is_truthy`/`is_falsey`; the decoder accepts those too, so
    /// a hand-written `--flash-attn 1` reads as the state it runs.
    pub fn parse(s: &str) -> Option<FlashAttn> {
        match s {
            "auto" => Some(FlashAttn::Auto),
            "on" | "1" | "true" | "yes" => Some(FlashAttn::On),
            "off" | "0" | "false" | "no" => Some(FlashAttn::Off),
            _ => None,
        }
    }
}

/// A backend's cache configuration: the K and V cache types and
/// the flash-attention mode, as DECODED from its raw flags. An
/// absent token decodes to the llama.cpp default for that
/// setting (f16 / f16 / auto) — the state the server runs — so
/// the decoded selection is always the effective one. Whether
/// the tokens are present at all (the state a pre-feature backend
/// is in, whose spawned command line must not change) is
/// [`CacheSelection::is_default`]-independent: the writer decides
/// presence, see [`set_cache_selection`].
#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
pub struct CacheSelection {
    pub k: CacheType,
    pub v: CacheType,
    pub flash_attn: FlashAttn,
}

impl Default for CacheSelection {
    /// llama.cpp's defaults: f16 K and V caches, flash attention
    /// auto (the pinned build's `common_params` default).
    fn default() -> Self {
        CacheSelection {
            k: CacheType::F16,
            v: CacheType::F16,
            flash_attn: FlashAttn::Auto,
        }
    }
}

impl CacheSelection {
    /// The combination llama.cpp refuses at context creation: a
    /// quantized V cache with flash attention explicitly off
    /// (auto would be promoted to enabled; off is a hard error —
    /// see the module docs). The save path refuses this pairing
    /// rather than persisting a backend that cannot start.
    pub fn v_quantized_fa_off(&self) -> bool {
        self.v.is_quantized() && self.flash_attn == FlashAttn::Off
    }
}

/// The value token of a `--flag value` pair, for flags whose
/// value is a word (never begins with `-`): the next token when
/// it exists and is not itself a flag. Mirrors [`super::mtp`]'s
/// helper of the same name.
fn pair_value<'a>(tokens: &[&'a str], i: usize) -> Option<&'a str> {
    tokens.get(i + 1).copied().filter(|v| !v.starts_with('-'))
}

/// The last value written for one of the cache flags, across the
/// pair and `=` spellings (llama.cpp's sequential-parse rule: the
/// last occurrence wins). `None` when the flag never appears with
/// a value.
fn last_flag_value<'a>(tokens: &[&'a str], flag: &str, short: &str) -> Option<&'a str> {
    let mut value: Option<&str> = None;
    let mut i = 0;
    while i < tokens.len() {
        let t = tokens[i];
        if t == flag || t == short {
            if let Some(v) = pair_value(tokens, i) {
                value = Some(v);
                i += 1;
            }
        } else if let Some(v) = t
            .strip_prefix(&format!("{flag}="))
            .or_else(|| t.strip_prefix(&format!("{short}=")))
        {
            value = Some(v);
        }
        i += 1;
    }
    value
}

/// Detect a backend's cache selection in raw llama-server flags.
///
/// Each setting resolves independently: the last parseable value
/// wins; an absent or unparseable token (a type v1 does not
/// offer, a malformed spelling) reads as the llama.cpp default
/// for that setting, so decoding never invents a setting the
/// flags do not effectively state.
pub fn cache_selection(raw_flags: Option<&str>) -> CacheSelection {
    let Some(raw) = raw_flags else {
        return CacheSelection::default();
    };
    let tokens: Vec<&str> = raw.split_whitespace().collect();
    let mut sel = CacheSelection::default();
    if let Some(t) = last_flag_value(&tokens, "--cache-type-k", "-ctk")
        .and_then(CacheType::parse)
    {
        sel.k = t;
    }
    if let Some(t) = last_flag_value(&tokens, "--cache-type-v", "-ctv")
        .and_then(CacheType::parse)
    {
        sel.v = t;
    }
    if let Some(f) = last_flag_value(&tokens, "--flash-attn", "-fa")
        .and_then(FlashAttn::parse)
    {
        sel.flash_attn = f;
    }
    sel
}

/// Whether any of the cache tokens is present in raw flags —
/// the distinction between a pre-feature backend (no tokens; its
/// spawned command line must stay exactly as it is until the
/// operator touches these controls) and one the controls have
/// written. Presence is about tokens, not values: a hand-written
/// `--cache-type-k f16` is present.
pub fn cache_tokens_present(raw_flags: Option<&str>) -> bool {
    let Some(raw) = raw_flags else {
        return false;
    };
    raw.split_whitespace().any(|t| {
        let name = t.split('=').next().unwrap_or(t);
        matches!(
            name,
            "--cache-type-k" | "-ctk" | "--cache-type-v" | "-ctv" | "--flash-attn" | "-fa"
        )
    })
}

/// Rewrite the cache tokens in raw llama-server flags, leaving
/// every other flag untouched and in order. All existing
/// `--cache-type-k` / `-ctk`, `--cache-type-v` / `-ctv`, and
/// `--flash-attn` / `-fa` occurrences (pair and `=` spellings)
/// are removed, then the canonical tokens are appended.
///
/// `None` removes the tokens without appending, returning the
/// backend to having no cache tokens at all (llama.cpp's
/// defaults, and the pre-feature state). A `Some` selection
/// always writes all three tokens explicitly — the state the app
/// wrote is the state the flags state, with no implicit member.
/// Returns `None` when nothing remains.
pub fn set_cache_selection(
    raw_flags: Option<&str>,
    setting: Option<CacheSelection>,
) -> Option<String> {
    let tokens: Vec<&str> = raw_flags.unwrap_or("").split_whitespace().collect();
    let owned = |t: &str| -> bool {
        let name = t.split('=').next().unwrap_or(t);
        matches!(
            name,
            "--cache-type-k" | "-ctk" | "--cache-type-v" | "-ctv" | "--flash-attn" | "-fa"
        )
    };
    let mut out: Vec<String> = Vec::new();
    let mut i = 0;
    while i < tokens.len() {
        let t = tokens[i];
        if owned(t) {
            // Skip the value token too when the flag is in pair
            // spelling and a value follows that is not itself a
            // flag; a malformed value token is left for the main
            // walk (mirrors set_parallel_slots).
            let has_value = !t.contains('=')
                && tokens
                    .get(i + 1)
                    .map(|v| !v.starts_with('-'))
                    .unwrap_or(false);
            i += if has_value { 2 } else { 1 };
            continue;
        }
        out.push(t.to_string());
        i += 1;
    }
    if let Some(sel) = setting {
        out.push("--cache-type-k".to_string());
        out.push(sel.k.as_str().to_string());
        out.push("--cache-type-v".to_string());
        out.push(sel.v.as_str().to_string());
        out.push("--flash-attn".to_string());
        out.push(sel.flash_attn.as_str().to_string());
    }
    if out.is_empty() {
        None
    } else {
        Some(out.join(" "))
    }
}

/// The headroom reserve in the predicted-best fit test: a rung
/// fits only when its predicted per-GPU total is at most
/// `100 - HEADROOM_RESERVE_PERCENT` percent of that GPU's total
/// VRAM. The requirement model prices weights, KV, and the
/// recurrent-state cache; llama.cpp additionally allocates
/// compute buffers and graph scratch the model does not price,
/// and Pete's Oct 20, 2026 measurements put a 64K f16 planner
/// configuration whose priced total was ~96.5% of the card in
/// the spill regime (13.2 tok/s vs 29.5 at 32K) — a 3% reserve
/// would call that configuration fitting. 5% is the smallest
/// round reserve that prices it as not fitting.
pub const HEADROOM_RESERVE_PERCENT: u64 = 5;

/// One rung of the predicted-best quality ladder: a cache
/// selection plus its predicted footprint.
#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
pub struct CacheRung {
    pub selection: CacheSelection,
    /// KV cache MiB at this rung's cache types.
    pub kv_mib: u64,
    /// Total predicted GPU-resident MiB (weights + KV +
    /// recurrent-state cache when its geometry is known).
    pub total_mib: u64,
    /// Whether the rung fits every target GPU under the headroom
    /// reserve (see [`HEADROOM_RESERVE_PERCENT`]).
    pub fits: bool,
}

/// The inputs the predicted-best chooser prices, resolved by the
/// caller from the backend's layout and placement: everything is
/// a number the existing requirement model already computes, so
/// the chooser is a pure function of them.
#[derive(Debug, Clone, PartialEq)]
pub struct CacheFitInputs {
    /// KV geometry of the backend's model.
    pub kv: KvGeometry,
    /// GPU-resident weight MiB under the backend's current
    /// offload configuration (`MemoryRequirement::gpu_weights_mib`).
    pub gpu_weights_mib: u64,
    /// Recurrent-state cache MiB at the effective slot count,
    /// when the model's geometry is known.
    pub rs_mib: Option<u64>,
    /// MoE expert cache MiB (`--moe-cache-mib`) the backend
    /// reserves: a constant addend to every rung's total (it
    /// does not vary with the cache selection being priced).
    /// 0 for backends that set no expert cache.
    pub moe_cache_mib: u64,
    /// The context size the KV cache is priced at, in tokens.
    /// This is the per-slot context the requirement model prices
    /// (llama.cpp's unified KV cache holds the full context once,
    /// shared across slots, at the pinned build's default).
    pub n_ctx: u32,
    /// The tensor split distributing the total over the target
    /// GPUs, in target order; empty for an unsplit placement (the
    /// whole total on the first target).
    pub split: Vec<f32>,
    /// Total VRAM MiB of each target GPU, in split order.
    pub gpu_total_mib: Vec<u64>,
}

/// KV bytes for one token at a selection: the K side priced with

/// The quality ladder, highest first (Pete, Oct 20, 2026:
/// predicted high-quality options with a balance of speed —
/// quality decides, and flash attention stays on at every rung:
/// on supported builds it is a pure speed/memory-traffic win at

/// degrades before K: the mixed rung quantizes V to q4_0 while K

pub const PREDICTED_BEST_LADDER: [CacheSelection; 4] = [

        k: CacheType::F16,
        v: CacheType::F16,
        flash_attn: FlashAttn::On,

    CacheSelection {

/// The per-GPU shares of a predicted total, distributed over
/// the target GPUs by the split (an empty split places
/// everything on the first target; the last target takes the
/// rounding remainder). `None` when there is nothing to
/// distribute over (no targets) or the split cannot distribute
/// (its length differs from the target count — the placement
/// projection's stale-split rule). This is the distribution
/// [`predicted_fits`] judges; the cache preview payload carries
/// the same shares as each target's required figure.
pub fn predicted_shares(
    total_mib: u64,
    split: &[f32],
    gpu_total_mib: &[u64],
) -> Option<Vec<u64>> {
    if gpu_total_mib.is_empty() {
        return None;
    }
    if split.is_empty() {
        let mut v = vec![0u64; gpu_total_mib.len()];
        v[0] = total_mib;
        return Some(v);
    }
    if split.len() != gpu_total_mib.len() {
        return None;
    }
    let mut assigned = 0u64;
    Some(
        split
            .iter()
            .enumerate()
            .map(|(i, frac)| {
                let share = if i + 1 == split.len() {
                    total_mib.saturating_sub(assigned)
                } else {
                    (total_mib as f64 * (*frac as f64)).round() as u64
                };
                assigned += share;
                share
            })
            .collect(),
    )
}

/// Whether a predicted total fits the target GPUs: every GPU's
/// share of the total (see [`predicted_shares`]) must be within
/// the headroom reserve of its total. With no targets there is
/// nothing to fit against.
pub fn predicted_fits(total_mib: u64, split: &[f32], gpu_total_mib: &[u64]) -> bool {
    match predicted_shares(total_mib, split, gpu_total_mib) {
        Some(shares) => shares.iter().zip(gpu_total_mib.iter()).all(|(share, total)| {
            *share <= total * (100 - HEADROOM_RESERVE_PERCENT) / 100
        }),
        None => false,
    }
}

/// Price every ladder rung against the inputs, in ladder order.
pub fn price_ladder(inputs: &CacheFitInputs) -> Vec<CacheRung> {
    PREDICTED_BEST_LADDER
        .iter()
        .map(|selection| {
            let kv_bytes = kv_bytes_per_token(&inputs.kv, selection)
                .saturating_mul(inputs.n_ctx as u64);
            let kv_mib = kv_bytes.div_ceil(MIB);
            let total_mib = inputs
                .gpu_weights_mib
                .saturating_add(kv_mib)
                .saturating_add(inputs.rs_mib.unwrap_or(0))
                .saturating_add(inputs.moe_cache_mib);
            CacheRung {
                selection: *selection,
                kv_mib,
                total_mib,
                fits: predicted_fits(total_mib, &inputs.split, &inputs.gpu_total_mib),
            }
        })
        .collect()
}

/// The predicted-best selection: the highest-quality rung that
/// fits, or `None` when no rung fits — the caller then reports
/// the lowest rung's predicted numbers and states the
/// configuration does not fit at the chosen context; it does not
/// silently reduce the context, change the placement, or apply
/// a non-fitting combination.
pub fn predicted_best(inputs: &CacheFitInputs) -> Option<CacheRung> {
    price_ladder(inputs).into_iter().find(|r| r.fits)
}

    #[test]
    fn block_geometry_is_the_ggml_quanta() {
        // ggml-common.h at the pinned build: block_q8_0 is an fp16
        // scale + 32 int8 quants = 34 B per 32 elements; block_q4_0
        // is an fp16 scale + 32 nibbles = 18 B per 32 elements.
        assert_eq!(CacheType::Q8_0.block_geometry(), (32, 34));
        assert_eq!(CacheType::Q4_0.block_geometry(), (32, 18));
        assert_eq!(CacheType::F16.block_geometry(), (1, 2));

        assert_eq!(CacheType::Q8_0.row_bytes(40), 68);

    /// Pete's Oct 20, 2026 measured case: Qwen3.8-27B Planner-MTP
    /// IQ3_S on the RTX A4000 (16 GiB), parallel 1. The geometry
    /// is the model's own (GGUF metadata shape: 64 blocks, full
    /// attention every 4th layer → 16 KV layers, 4 KV heads, head
    /// dimensions 256/256); the file is the catalog pin
    /// (12,120,016,960 bytes including the MTP head); the
    /// recurrent-state term is the family geometry the GGUF parse
    /// yields (156,893,184 B per sequence — see the planner-moe
    /// requirement test in memory.rs), at one slot.
    fn planner_mtp_inputs(n_ctx: u32) -> CacheFitInputs {
        let req = super::super::memory::memory_requirement(
            12_120_016_960,
            Some(0),
            None,
            super::super::MemoryBasis::Catalog,
            super::super::ExpertOffload::None,
            Some(65_536),
            None,
            n_ctx,
            Some(156_893_184),
            1,
        );
        CacheFitInputs {
            kv: KvGeometry {
                kv_layers: 16,
                head_count_kv: 4,
                key_length: 256,
                value_length: 256,
            },
            gpu_weights_mib: req.gpu_weights_mib,
            rs_mib: req.rs_mib,
            moe_cache_mib: 0,
            n_ctx,
            split: Vec::new(),
            gpu_total_mib: vec![16_384],
        }
    }

    #[test]
    fn chooser_reproduces_petes_measured_cases() {
        // 32K: rung 1 (f16/f16, FA on) fits — his 29.5 tok/s point.
        let inputs = planner_mtp_inputs(32_768);
        let ladder = price_ladder(&inputs);
        assert_eq!(ladder[0].kv_mib, 2_048);
        assert_eq!(ladder[0].total_mib, 11_559 + 2_048 + 150);
        assert!(ladder[0].fits);
        assert_eq!(predicted_best(&inputs).unwrap(), ladder[0]);

        // 64K: f16 does not fit — his measured spill regime
        // (13.2 tok/s) — and the q8_0 rung is chosen (his q8_0
        // recovery measurement point).
        let inputs = planner_mtp_inputs(65_536);
        let ladder = price_ladder(&inputs);
        assert_eq!(ladder[0].kv_mib, 4_096);
        assert_eq!(ladder[0].total_mib, 11_559 + 4_096 + 150);
        assert!(!ladder[0].fits, "64K f16 must not fit: {:?}", ladder[0]);
        assert_eq!(ladder[1].kv_mib, 2_176);
        assert!(ladder[1].fits);
        let best = predicted_best(&inputs).unwrap();
        assert_eq!(best.selection.k, CacheType::Q8_0);
        assert_eq!(best.selection.v, CacheType::Q8_0);
        assert_eq!(best.selection.flash_attn, FlashAttn::On);

        // 256K: no rung fits at the chosen context; the lowest
        // rung's numbers are the honest report (16,317 MiB against
        // a 16,384 MiB card — under the raw total, over the
        // headroom limit, so still no fit claim).
        let inputs = planner_mtp_inputs(262_144);
        let ladder = price_ladder(&inputs);
        assert!(ladder.iter().all(|r| !r.fits));
        assert_eq!(predicted_best(&inputs), None);
        let lowest = ladder[3];
        assert_eq!(lowest.kv_mib, 4_608);
        assert_eq!(lowest.total_mib, 11_559 + 4_608 + 150);
        assert!(lowest.total_mib < 16_384);
    }

    #[test]
    fn ladder_degrades_v_before_k() {
        assert_eq!(PREDICTED_BEST_LADDER[2].k, CacheType::Q8_0);
        assert_eq!(PREDICTED_BEST_LADDER[2].v, CacheType::Q4_0);
        // Flash attention stays on at every rung.
        assert!(PREDICTED_BEST_LADDER
            .iter()
            .all(|s| s.flash_attn == FlashAttn::On));
    }

    #[test]
    fn predicted_shares_distribute_like_predicted_fits() {
        // Unsplit: the whole total lands on the first target.
        assert_eq!(predicted_shares(1_000, &[], &[16_384]), Some(vec![1_000]));
        assert_eq!(
            predicted_shares(1_000, &[], &[16_384, 16_384]),
            Some(vec![1_000, 0])
        );
        // Split: shares follow the fractions and the last target
        // takes the rounding remainder, so shares sum to the total.
        assert_eq!(
            predicted_shares(10_001, &[0.5, 0.5], &[16_384, 16_384]),
            Some(vec![5_001, 5_000])
        );
        assert_eq!(
            predicted_shares(900, &[0.25, 0.25, 0.5], &[16_384, 16_384, 16_384]),
            Some(vec![225, 225, 450])
        );
        // Nothing to distribute over, and a stale split: no shares.
        assert_eq!(predicted_shares(1_000, &[], &[]), None);
        assert_eq!(predicted_shares(1_000, &[1.0], &[16_384, 16_384]), None);
    }

    #[test]
    fn split_placement_fits_per_gpu() {
        // The same 64K f16 total split evenly across two 16 GiB
        // GPUs fits each card's share, though it fits neither
        // card alone.
        let mut inputs = planner_mtp_inputs(65_536);
        inputs.split = vec![0.5, 0.5];
        inputs.gpu_total_mib = vec![16_384, 16_384];
        assert!(price_ladder(&inputs)[0].fits);
        // A stale split (length mismatch) distributes nothing and
        // fits nothing.
        inputs.split = vec![1.0];
        assert!(!price_ladder(&inputs)[0].fits);
        // No targets: no fit claim.
        inputs.split = Vec::new();
        inputs.gpu_total_mib = Vec::new();
        assert!(!price_ladder(&inputs)[0].fits);
    }
